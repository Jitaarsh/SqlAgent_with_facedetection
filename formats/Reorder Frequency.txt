SELECT SUBSTR(m.<date_col>, 1, 4) AS Year, c.<id_col> AS CustomerID,
       SUBSTRING_INDEX(c.<name_col>, ',', 1) AS CustomerName,
       p.<name_col> AS ProductName, COUNT(*) AS Frequency, ROUND(SUM(m.<cases_col>), 4) AS Total
FROM month_data m
JOIN customers c ON m.<customer_fk> = c.<customer_pk>
JOIN products p ON m.<product_fk> = p.<product_pk>
GROUP BY SUBSTR(m.<date_col>, 1, 4), c.<id_col>, c.<name_col>, p.<name_col>
HAVING ROUND(SUM(m.<cases_col>), 4) > 0
ORDER BY CustomerName, Year;
