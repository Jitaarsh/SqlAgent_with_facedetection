EXAMPLE QUERY PATTERN (use real column names from describe_table)

SELECT
    m.<date_col>                    AS YearMonth,
    c.<name_col>                    AS CustomerName,
    p.<name_col>                    AS ProductName,
    ROUND(SUM(m.<cases_col>), 4)    AS Total
FROM month_data m
JOIN customers c ON m.<customer_fk> = c.<customer_pk>
JOIN products p ON m.<product_fk> = p.<product_pk>
WHERE m.<customer_fk> NOT IN (
    SELECT DISTINCT m2.<customer_fk> FROM month_data m2
    WHERE m2.<date_col> < m.<date_col>
)
GROUP BY m.<date_col>, c.<name_col>, p.<name_col>
HAVING ROUND(SUM(m.<cases_col>), 4) > 0
ORDER BY YearMonth, CustomerName;
